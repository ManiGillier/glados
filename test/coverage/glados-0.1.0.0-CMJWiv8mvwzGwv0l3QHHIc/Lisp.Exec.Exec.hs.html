   18 -- Main function of lisp execution
   19 -- Return result and modified String table 
   20 execLisp :: Ast -> SymTable -> (MaybeError String, SymTable)
   21 execLisp ast env = 
   22     let (val, e) = evalAst env ast     
   23     in case val of
   24         Just (VInt v) -> (Correct (show v), e)
   25         Just (VBool v) -> (Correct $ boolToSymbol v, e)
   26         Just (VLambda _ _ _) -> ((Error (Error.procError ++ ">") []), e) 
   27         Just (VError s) -> (Error s [], e)
   28         Nothing ->  (Correct [],e)
   29 
   30 -- Eval case of Ast
   31 evalAst :: SymTable -> Ast -> (Maybe Value, SymTable)
   32 evalAst env ast =
   33     case ast of
   34         Value x -> (Just (VInt x), env)
   35         Boolean x -> (Just (VBool x), env)
   36         Symbol s -> lookupSymbol env s
   37         Define sym a -> defineSymbol env sym a
   38         Call f args -> evalCall env f args
   39         Lambda param body -> evalLambda env param body
   40         Apply lambdaExpr args -> evalApply env lambdaExpr args
   41         If cond th el -> evalCond env cond th el
   42 
   43 -- Eval if -> then -> else
   44 evalCond :: SymTable -> Ast -> Ast -> Ast -> (Maybe Value, SymTable)
   45 evalCond env conditionAst thenBranch elseBranch = 
   46     let (maybeCondition, newEnv) = evalAst env conditionAst
   47     in case maybeCondition of
   48         Just (VBool True) -> evalAst newEnv thenBranch
   49         Just (VBool False) -> evalAst newEnv elseBranch
   50         Just (VInt 0) -> evalAst newEnv elseBranch
   51         Just (VInt _) -> evalAst newEnv thenBranch
   52         Just (VError err) -> (Just (VError err), newEnv)
   53         Nothing -> evalAst newEnv elseBranch
   54         _ -> (Just (VError Error.condError), newEnv)
   55 
   56 -- Check if the String already exist
   57 lookupSymbol :: SymTable -> String -> (Maybe Value, SymTable)
   58 lookupSymbol [] var = (Just (VError (Error.notBoundError var)), [])
   59 lookupSymbol env@((sym, val):rest) target
   60     | sym == target = 
   61         case val of
   62             VLambda _ _ _ -> 
   63                 (Just (VError (Error.procError ++ " " ++ target ++ ">")), env)
   64             _             -> (Just val, env)
   65     | otherwise =
   66         let (res, _) = lookupSymbol rest target
   67         in (res, env)
   68 
   69 -- Define String and keep it with env variable 
   70 defineSymbol :: SymTable -> String -> Ast -> (Maybe Value, SymTable)
   71 defineSymbol env s a = 
   72     let (maybeVal, newEnv) = evalAst env a
   73     in case maybeVal of
   74         Just val -> (Nothing, updateEnv newEnv s val)
   75         Nothing -> (Nothing, env)
   76 
   77 updateEnv :: SymTable -> String -> Value -> SymTable
   78 updateEnv [] s val = [(s, val)]
   79 updateEnv ((sym, oldVal):rest) s val
   80     | sym == s = (sym, val) : rest
   81     | otherwise = (sym, oldVal) : updateEnv rest s val
   82 
   83 -- Call buitlin or defined lambda
   84 evalCall :: SymTable -> String -> [Ast] -> (Maybe Value, SymTable)
   85 evalCall env fName args 
   86     | isBuiltin fName && (not $ isDefined fName env) 
   87         = evalBuiltinCall env fName args
   88     | otherwise = evalUserCall env fName args
   89 
   90 isBuiltin :: String -> Bool
   91 isBuiltin s = s `elem` ["+", "-", "*", "div", "mod", "eq?", "<"]
   92 
   93 isVariadic :: String -> Bool
   94 isVariadic s = s `elem` ["+", "-", "*"]
   95 
   96 isDefined :: String -> SymTable -> Bool
   97 isDefined s env = s `elem` map fst env
   98 
   99 boolToSymbol :: Bool -> String
  100 boolToSymbol x
  101     | x = "#t"
  102     | otherwise = "#f"
  103 
  104 evalBuiltinCall :: SymTable -> String -> [Ast] -> (Maybe Value, SymTable)
  105 evalBuiltinCall env fName args 
  106     | not (isVariadic fName) && (length args > 2) = 
  107         (Just (VError $ Error.argsError args 
  108         ++ " " ++Error.callError fName args), env)
  109     | otherwise = callBuiltin env fName args
  110 
  111 callBuiltin :: SymTable -> String -> [Ast] -> (Maybe Value, SymTable)
  112 callBuiltin env fName args = case fName of
  113         "+" -> (fmap VInt (evalVariadicOp env (+) 0 args), env)
  114         "-" -> (fmap VInt (evalVariadicSub env args), env)
  115         "*" -> (fmap VInt (evalVariadicOp env (*) 1 args), env)
  116         "div" -> (fmap VInt (evalBinaryOp env Builtin.safeDiv args), env)
  117         "mod" -> (fmap VInt (evalBinaryOp env Builtin.safeMod args), env)
  118         "eq?" -> (fmap VBool (iToB (evalBinaryOp env Builtin.equal args)), env)
  119         "<" -> (fmap VBool (iToB (evalBinaryOp env Builtin.infsign args)), env)
  120         _ -> (Nothing, env)
  121 
  122 evalVariadicOp :: SymTable -> (Int -> Int -> Int) -> Int -> [Ast] -> Maybe Int
  123 evalVariadicOp env op identity args = 
  124     fmap (foldl op identity) (mapM (evalToInt env) args)
  125 
  126 evalVariadicSub :: SymTable -> [Ast] -> Maybe Int
  127 evalVariadicSub _ [] = Nothing
  128 evalVariadicSub env [arg] = 
  129     case evalToInt env arg of
  130         Just value -> Just (-value)
  131         Nothing -> Nothing
  132 evalVariadicSub env (first:rest) = 
  133     case evalToInt env first of
  134         Just firstVal -> 
  135             case mapM (evalToInt env) rest of
  136                 Just restVals -> Just (foldl (-) firstVal restVals)
  137                 Nothing -> Nothing
  138         Nothing -> Nothing
  139 
  140 evalToInt :: SymTable -> Ast -> Maybe Int
  141 evalToInt env ast = 
  142     case fst (evalAst env ast) of
  143         Just (VInt v) -> Just v
  144         _ -> Nothing
  145 
  146 -- Evaluate user-defined functions (lambdas)
  147 evalUserCall :: SymTable -> String -> [Ast] -> (Maybe Value, SymTable)
  148 evalUserCall env fName args = 
  149     case lookup fName env of
  150         Just (VLambda para body clEnv) -> 
  151             applyLambda env para body clEnv args fName
  152         Just (VInt x) -> (Just (VError (Error.nonProcedError (Just x))), env)
  153         Just (VBool _) -> err 
  154         Just (VError _) -> err
  155         Nothing -> err
  156         where 
  157             err = (Just (VError (Error.notBoundError fName)), env)
  158 
  159 evalBinaryOp :: SymTable -> (Int -> Int -> Int) -> [Ast] -> Maybe Int
  160 evalBinaryOp env op [arg1, arg2] =
  161     case (fst (evalAst env arg1), fst (evalAst env arg2)) of
  162         (Just (VInt v1), Just (VInt v2)) -> Just (op v1 v2)
  163         _                                -> Nothing
  164 evalBinaryOp _ _ _ = Nothing
  165 
  166 -- Create a lambda
  167 evalLambda :: SymTable -> [String] -> Ast -> (Maybe Value, SymTable)
  168 evalLambda env param body = (Just (VLambda param body env), env)
  169 
  170 -- Apply lambda
  171 applyLambda :: SymTable -> [String] -> Ast -> SymTable -> [Ast] -> String -> (Maybe Value, SymTable)
  172 applyLambda env param body clEnv args funcName
  173     | length param /= length args = (Just (VError (Error.argsError args)), env)
  174     | otherwise = let argValues = map (fst . evalAst env) args
  175         in if all isJust argValues
  176            then let justValues = map fromJust argValues
  177                     recursiveEnv = (funcName, VLambda param body clEnv) : clEnv
  178                     localEnv = zip param justValues ++ recursiveEnv
  179                     (result, _) = evalAst localEnv body
  180                 in (result, env)
  181            else (Nothing, env)
  182 
  183 -- Apply anonymous Lambda call ex: ((lambda (a b c) (* a (* b c))) 2 2 2)
  184 evalApply :: SymTable -> Ast -> [Ast] -> (Maybe Value, SymTable)
  185 evalApply env lambdaExpr args = 
  186     let (maybeLambda, newEnv) = evalAst env lambdaExpr
  187     in case maybeLambda of
  188         Just (VLambda param body clEnv) -> 
  189             applyLambdaAnonymous newEnv param body clEnv args
  190         _ -> (Nothing, newEnv)
  191 
  192 applyLambdaAnonymous :: SymTable -> [String] -> Ast -> SymTable -> [Ast] -> (Maybe Value, SymTable)
  193 applyLambdaAnonymous env param body clEnv args
  194     | length param /= length args = (Just (VError (Error.argsError args)), env)
  195     | otherwise = 
  196         let argValues = map (fst . evalAst env) args
  197         in if all isJust argValues
  198            then let justValues = map fromJust argValues
  199                     localEnv = zip param justValues ++ clEnv
  200                     (result, _) = evalAst localEnv body
  201                 in (result, env)
  202            else (Nothing, env)
