    8 module Error.MaybeError ( ErrorType, MaybeError (..), fmap, pure, (<*>), (>>=)
    9              , printError
   10              , toMaybeError
   11              , toMaybe
   12              , invertMaybe ) where
   13 
   14 import System.IO ( hPutStrLn, stderr )
   15 
   16 type ErrorType = String
   17 
   18 data MaybeError a = Error !ErrorType !String | Correct !a
   19   deriving (Eq)
   20 
   21 instance Show a => Show (MaybeError a) where
   22   show (Correct x) = show x
   23   show (Error t "") = t
   24   show (Error t s) = t ++ ": " ++ s
   25 
   26 instance Functor MaybeError where
   27   fmap f (Correct x) = Correct (f x)
   28   fmap _ (Error errorType str) = Error errorType str
   29 
   30 instance Applicative MaybeError where
   31   pure = Correct
   32 
   33   (<*>) (Error t s) _ = Error t s
   34   (<*>) _ (Error t s) = Error t s
   35   (<*>) (Correct f) (Correct x) = Correct (f x)
   36 
   37 instance Monad MaybeError where
   38   (>>=) (Correct x) f = f x
   39   (>>=) (Error errorType str) _ = Error errorType str
   40 
   41 toMaybeError :: Maybe a -> (ErrorType,String) -> MaybeError a
   42 toMaybeError Nothing (err_type,err_str) = Error err_type err_str
   43 toMaybeError (Just result) _ = Correct result
   44 
   45 printError :: MaybeError a -> IO ()
   46 printError (Error t c) = hPutStrLn stderr (t ++ ": " ++ c)
   47 printError _ = return ()
   48 
   49 toMaybe :: MaybeError a -> Maybe a
   50 toMaybe (Correct x) = Just x
   51 toMaybe (Error _ _) = Nothing
   52 
   53 invertMaybe :: Maybe (MaybeError a) -> MaybeError (Maybe a)
   54 invertMaybe Nothing = Correct Nothing
   55 invertMaybe (Just (Correct a)) = Correct (Just a)
   56 invertMaybe (Just (Error x y)) = Error x y
