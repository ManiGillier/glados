    8 module Lisp.Exec.ErrorExec (argsError,
    9                             nonProcedError,
   10                             notBoundError,
   11                             procError,
   12                             callError,
   13                             condError)
   14                             where
   15 argsError :: [a] -> String
   16 argsError args = "*** ERROR : wrong number of argument of " 
   17     ++ show (length args)
   18 
   19 nonProcedErrorStr :: String
   20 nonProcedErrorStr = "*** ERROR : attempt to apply non-procedure"
   21 
   22 nonProcedError :: Maybe Int -> String
   23 nonProcedError (Just x) = (nonProcedErrorStr ++ " " ++ show x)
   24 nonProcedError Nothing = nonProcedErrorStr 
   25 
   26 notBoundError :: String -> String
   27 notBoundError var = "*** ERROR : variable " ++ var ++ " is not bound"
   28 
   29 callError ::Show a => String -> [a] -> String
   30 callError fName args = 
   31     "in call (" ++ fName ++ concatMap ((" " ++) . show) args ++ ")"
   32 
   33 condError :: String
   34 condError = "*** ERROR : Invalid condition type"
   35 
   36 procError :: String
   37 procError = "#<procedure"
