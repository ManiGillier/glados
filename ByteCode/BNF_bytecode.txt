<bytecode> ::= <magic_number> <adress> <operation_list>

<magic_number> ::= "69,12,69,12,"

<adress> ::= <byte> "," <byte> "," <byte> "," <byte> "," <byte> "," <byte> "," <byte> "," <byte> [","]

<operation_list> ::= <operation> | <operation> "," <operation_list>

<operation> ::= <not>
              | <lnot>
              | <neg>
              | <and>
              | <or>
              | <land>
              | <lor>
              | <xor>
              | <shl>
              | <shr>
              | <add>
              | <sub>
              | <mult>
              | <div>
              | <mod>
              | <gt>
              | <ge>
              | <lt>
              | <le>
              | <eq>
              | <diff>
              | <updz>
              | <push_stack>
              | <pop_stack>
              | <pop>
              | <call>
              | <ret>
              | <jmp>
              | <zjmp>
              | <aff>
              | <affs>
              | <push_value>
              | <push_label>
